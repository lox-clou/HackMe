// Function to check the solution for a level
function checkSolution(title, input) {
    const level = levels.find(l => l.title === title);
    if (input.trim() === level.solution.toLowerCase()) {
        const container = document.querySelector(`.level[data-level="${title}"]`);
        container.classList.add('completed');
        container.style.animation = 'pulse 2s infinite';
        const hint = document.querySelector(`.hint[data-level="${title}"]`);
        hint.style.display = 'none';
        const flag = document.querySelector(`.flag[data-level="${title}"]`);
        flag.textContent = `FLAG{${level.title.replace(/\s+/g, '_').toUpperCase()}}`;
    } else {
        alert(`Incorrect solution for ${title}.`);
    }
}

// Add event listeners to update progress bar
document.querySelectorAll('.sidebar li').forEach(li => {
    li.addEventListener('click', () => {
        const level = document.querySelector(`.level[data-level="${li.dataset.level}"]`);
        if (level.classList.contains('completed')) {
            localStorage.setItem(`${li.dataset.level}_status`, 'completed');
        } else {
            localStorage.setItem(`${li.dataset.level}_status`, 'in_progress');
        }
    });
});
